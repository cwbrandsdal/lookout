cmake_minimum_required(VERSION 3.20)
project(LookoutHandoff LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Lookout handoff requires the Windows x64 MSVC toolchain")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
find_program(MIDL_EXECUTABLE midl REQUIRED)
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED}")
add_custom_command(
  OUTPUT "${GENERATED}/ITerminalHandoff.h" "${GENERATED}/ITerminalHandoff_i.c" "${GENERATED}/ITerminalHandoff_p.c" "${GENERATED}/dlldata.c"
  COMMAND "${MIDL_EXECUTABLE}" /nologo /env x64 /target NT100 /robust /out "${GENERATED}" /h ITerminalHandoff.h /iid ITerminalHandoff_i.c /proxy ITerminalHandoff_p.c /dlldata dlldata.c "${CMAKE_CURRENT_SOURCE_DIR}/ITerminalHandoff.idl"
  DEPENDS ITerminalHandoff.idl VERBATIM)
add_library(LookoutHandoffProxy SHARED "${GENERATED}/ITerminalHandoff_p.c" "${GENERATED}/ITerminalHandoff_i.c" "${GENERATED}/dlldata.c" proxy.def)
target_compile_options(LookoutHandoffProxy PRIVATE /FI${CMAKE_CURRENT_SOURCE_DIR}/proxy_clsid.h)
target_compile_definitions(LookoutHandoffProxy PRIVATE _WIN32_WINNT=0x0A00)
target_link_libraries(LookoutHandoffProxy PRIVATE rpcrt4 oleaut32 uuid)
foreach(target LookoutHandoff LookoutHandoffSelfTest)
  if(target STREQUAL "LookoutHandoff")
    add_executable(${target} WIN32 main.cpp "${GENERATED}/ITerminalHandoff_i.c")
  else()
    add_executable(${target} self-test.cpp "${GENERATED}/ITerminalHandoff_i.c")
  endif()
  target_include_directories(${target} PRIVATE "${GENERATED}")
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_compile_options(${target} PRIVATE /W4 /EHsc)
  target_link_libraries(${target} PRIVATE ole32 oleaut32 advapi32 uuid shell32)
  add_dependencies(${target} LookoutHandoffProxy)
endforeach()
